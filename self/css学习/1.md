    Extract --> VendorType{"供应商类型判定"}

    PayModeMS -- "License (先付后用)" --> BizFlowMS_Pre["生成采购订单"]
    
    %% 分支 B: 非微软业务
    VendorType -- "非微软业务" --> PayModeOther{"付费模式 (非微软)"}
    PayModeOther -- "先用后付" --> BizFlowOther_Post["导入账单数据"]
    PayModeOther -- "先付后用" --> BizFlowOther_Pre["生成采购订单 "]
    
    %% --- 3. 进销存业务执行 (业务层) ---
    BizFlowMS_Post --> Inbound["虚拟入库 (确认厂商成本)"]
    BizFlowMS_Pre --> Inbound
    BizFlowOther_Post --> Inbound
    BizFlowOther_Pre --> Inbound
    
    Inbound --> Outbound["虚拟出库 (确认客户消耗/交付)"]
    
    %% --- 4. 财务计划与核算生成 (核心修改点：移到此处) ---
    %% 这里体现“业财一体”：业务动作触发财务生成

    Discount --> CalcEngine

    GenARAP --> AR["客户账单: 发送给客户"]
    
    %% --- 5. 资金结算与核销 ---
    Bank["银行流水/支付记录"] --> Verify{"核销匹配"}
    AR --> Verify

    
    Verify -- "回款/付款成功" --> WriteOff["财务核销"]
    
    WriteOff --> Report["生成: 利润表"]
    Report --> End["流程结束"]
